Step 2 — `src/routes/cases/[slug]/+page.ts` loader. Build a slug→Picture map at module load and resolve via the existing frontmatter string:

```ts
import { error } from '@sveltejs/kit';
import type { PageLoad } from './$types';

export const prerender = true;

// Eager glob: all hero images for cases. Key is the absolute Vite-style path.
const caseHeroes = import.meta.glob<{ default: unknown }>(
  '/src/lib/assets/cases/**/hero.{png,jpg,jpeg,webp,avif}',
  { query: { enhanced: true }, eager: true }
);

// Index by slug (penultimate path segment).
const heroBySlug = new Map<string, unknown>();
for (const [path, mod] of Object.entries(caseHeroes)) {
  const parts = path.split('/');
  const slug = parts[parts.length - 2];
  heroBySlug.set(slug, (mod as { default: unknown }).default);
}

export function entries() {
  const modules = import.meta.glob('/src/content/cases/*.md', { eager: true });
  return Object.keys(modules).map((path) => ({
    slug: path.split('/').pop()!.replace('.md', '')
  }));
}

export const load: PageLoad = async ({ params }) => {
  try {
    const post = await import(`../../../content/cases/${params.slug}.md`);
    const heroPicture = heroBySlug.get(params.slug) ?? null;
    return {
      content: post.default,
      meta: post.metadata,
      heroPicture, // new: Picture object or null
      seo: {
        title: `${post.metadata.title} | HoneyLink`,
        description: post.metadata.excerpt,
        image: `https://honeylink.nl${post.metadata.image}`, // SEO keeps absolute URL
        type: 'article'
      }
    };
  } catch {
    error(404, `Case study niet gevonden: ${params.slug}`);
  }
};
```
