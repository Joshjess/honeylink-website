**Step 1: Create `src/routes/sitemap.xml/+server.ts`:**
```typescript
import type { RequestHandler } from '@sveltejs/kit';
import * as sitemap from 'super-sitemap';
import { getBlogPosts, getCases } from '$lib/server/content';

export const GET: RequestHandler = async () => {
  const blogSlugs = getBlogPosts().map((p) => p.slug);
  const caseSlugs = getCases().map((c) => c.slug);

  return await sitemap.response({
    origin: 'https://honeylink.nl',
    paramValues: {
      '/blogs/[slug]': blogSlugs,
      '/cases/[slug]': caseSlugs
    },
    excludeRoutePatterns: [
      '^/sitemap\\.xml$'
    ]
  });
};
```
Uses super-sitemap for auto route discovery per SEO-04. Passes all blog and case slugs via paramValues so parameterized routes are included per Research pitfall 7. Per SEO-01, this ensures all 36 existing URLs appear.
